  function scrollDown() {
    messages.scrollTop = messages.scrollHeight;
  }

  // Розміщує початок нової відповіді біля верху області чату.
  // Не прокручує всю сторінку магазину.
  function showAnswerStart(bubble) {
    function positionAnswer() {
      if (!chatWindow.classList.contains('open') || !bubble) return;
      var area = messages.getBoundingClientRect();
      var item = bubble.getBoundingClientRect();
      messages.scrollTop += item.top - area.top - 6;
    }
    if (window.requestAnimationFrame) {
      window.requestAnimationFrame(function () {
        positionAnswer();

      });
    } else {
      positionAnswer();
    }
  }

  function isMobileChat() {
    return window.matchMedia ? window.matchMedia('(max-width: 480px)').matches : window.innerWidth <= 480;
  }

  // На мобільних Safari/Chrome клавіатура змінює visual viewport.
  // Рахуємо координати від видимої області, а не від старого 78vh.
  function sizeMobileChat() {
    if (!isMobileChat()) {
      chatWindow.style.top = '';
      chatWindow.style.left = '';
      chatWindow.style.width = '';
      chatWindow.style.height = '';
      return;
    }
    var viewport = window.visualViewport;
    var top = viewport ? viewport.offsetTop : 0;
    var left = viewport ? viewport.offsetLeft : 0;
    var width = viewport ? viewport.width : window.innerWidth;
    var height = viewport ? viewport.height : window.innerHeight;
    var pad = 8;
    chatWindow.style.top = Math.round(top + pad) + 'px';
    chatWindow.style.left = Math.round(left + pad) + 'px';
    chatWindow.style.width = Math.max(0, Math.floor(width - pad * 2)) + 'px';
    chatWindow.style.height = Math.max(200, Math.floor(height - pad * 2)) + 'px';
  }

  function onViewportChange() {
    if (chatWindow.classList.contains('open')) sizeMobileChat();
  }
  if (window.visualViewport) {
    window.visualViewport.addEventListener('resize', onViewportChange);
    window.visualViewport.addEventListener('scroll', onViewportChange);
  }
  window.addEventListener('resize', onViewportChange);
  window.addEventListener('orientationchange', onViewportChange);

  // Accept only genuine HTTPS addresses from the Zorro catalog.
  // No redirects to supplier sites or AI-invented addresses.
  function safeZorroUrl(raw) {
    var value = String(raw || '');
    if (!/^https:\/\//i.test(value) || value.length > 2048 || /[\x00-\x20\x7f]/.test(value)) return '';
    var check = document.createElement('a');
    check.href = value;
    if (check.protocol !== 'https:' ||
        !/^(www\.)?zorro\.com\.ua$/i.test(check.hostname) ||
        check.username || check.password ||
        (check.port && check.port !== '443')) return '';
    return check.href;
  }

  // Product cards from the existing API also contain catalog-verified URLs.
  // This keeps text links working with older API responses during rollout.
  function collectProductLinks(links, cards) {
    var result = [];
    var seen = {};
    function include(entry) {
      if (!entry || typeof entry.name !== 'string' || !safeZorroUrl(entry.url)) return;
      var key = entry.name.toLowerCase();
      if (seen[key]) return;
      seen[key] = true;
      result.push({name: entry.name, url: entry.url});
    }
    if (Array.isArray(links)) links.forEach(include);
    if (Array.isArray(cards)) cards.forEach(include);
    return result.slice(0, 20);
  }

  // Use verified catalog links from the current turn AND earlier card responses.
  // Helpful when the user asks to describe or compare a previously shown model.
  function knownProductLinks(links, cards) {
    var collected = collectProductLinks(links, cards);
    var index;
    for (index = history.length - 1; index >= 0; index--) {
      var turn = history[index];
      var more = collectProductLinks(turn.product_links, turn.products);
      var i;
      for (i = 0; i < more.length && collected.length < 40; i++) {
        var present = false;
        var j;
        for (j = 0; j < collected.length; j++) {
          if (collected[j].name.toLowerCase() === more[i].name.toLowerCase()) {
            present = true;
            break;
          }
        }
        if (!present) collected.push(more[i]);
      }
    }
    return collected;
  }

  // Build DOM text nodes and anchors. NEVER set innerHTML from Gemini text.
  function linkedProductText(raw, links) {
    var value = String(raw || '');
    var approved = [];
    if (Array.isArray(links)) {
      links.forEach(function (entry) {
        if (!entry || typeof entry.name !== 'string' ||
            entry.name.length < 5 || entry.name.length > 180) return;
        var url = safeZorroUrl(entry.url);
        if (!url) return;
        approved.push({name: entry.name, lower: entry.name.toLowerCase(), url: url});
      });
    }
    approved.sort(function (a, b) { return b.name.length - a.name.length; });
    approved = approved.slice(0, 15);
    var frag = document.createDocumentFragment();
    var lower = value.toLowerCase();
    var cursor = 0;
    var parts = 0;
    while (cursor < value.length && parts < 100) {
      var winner = null;
      var where = value.length;
      approved.forEach(function (entry) {
        var index = lower.indexOf(entry.lower, cursor);
        if (index !== -1 && index < where) {
          winner = entry;
          where = index;
        }
      });
      if (!winner) break;
      if (where > cursor) frag.appendChild(document.createTextNode(value.slice(cursor, where)));
      var a = create('a', 'zorro-chat-inline-link', value.slice(where, where + winner.name.length));
      a.href = winner.url;
      a.target = '_self';
      a.rel = 'noopener noreferrer';
      frag.appendChild(a);
      cursor = where + winner.name.length;
      parts++;
    }
    if (cursor < value.length) frag.appendChild(document.createTextNode(value.slice(cursor)));
    return frag;
  }

  function addBubble(text, user, error, productLinks, checkout) {
    var className = 'zorro-chat-bubble ';

    if (user) {
      className += 'zorro-chat-user';
    } else {
      className += 'zorro-chat-bot';
    }

    if (error) {
      className += ' zorro-chat-error';
    }

    // Keep previous text-only rendering when no link metadata is supplied.
    var bubble = create('div', className);
    if (!user && !error && Array.isArray(productLinks) && productLinks.length) {
      bubble.appendChild(linkedProductText(text, productLinks));
    } else {
      bubble.textContent = String(text);
    }

    // A direct visit to the verified product page. Not an API order or a cart mutation.
    if (!user && !error && checkout && safeZorroUrl(checkout.url)) {
      bubble.appendChild(document.createTextNode('\n'));
      var checkoutButton = create('a', 'zorro-chat-checkout-link', 'Перейти до товару');
      checkoutButton.href = safeZorroUrl(checkout.url);
      checkoutButton.target = '_self';
      checkoutButton.rel = 'noopener noreferrer';
      bubble.appendChild(checkoutButton);
    }

    messages.appendChild(bubble);
    if (user) scrollDown();
    return bubble;
  }

  // Questions are derived from actual documented attribute labels; no claims
  // about product quality, water suitability, or warranty are assumed.
  function makeWatchQuestions(product) {
    var attributes = Array.isArray(product.attributes) ? product.attributes : [];
    var names = attributes.map(function (item) {
      return String(item && item.name || '').toLowerCase();
    }).join(' ');
    var items = [];
    function add(label, text) {
      var i;
      for (i = 0; i < items.length; i++) if (items[i].label === label) return;
      items.push({label: label, text: text});
    }
    var n = String(product.name || 'цього годинника');
    if (/вод|water|атм|atm|^ip\d/.test(names))
      add('Який захист від води?', 'Який підтверджений захист від води у ' + n + '? Чи можна його мочити?');
    if (/розмір|діаметр|габарит|товщин|вага/.test(names))
      add('Який розмір?', 'Які розміри і вага годинника ' + n + ' за характеристиками?');
    if (/ремін|браслет|матеріал/.test(names))
      add('Який ремінець?', 'Який матеріал ремінця годинника ' + n + '?');
    if (/заряд|батаре|акумул|автоном/.test(names))
      add('Як довго працює?', 'Яка підтверджена автономність годинника ' + n + '?');
    if (/дзвін|розмов|bluetooth|sim|сім/.test(names))
      add('Чи можна телефонувати?', 'Чи підтримує годинник ' + n + ' телефонні дзвінки?');
    if (/механізм|кварц/.test(names))
      add('Який механізм?', 'Який механізм у годинника ' + n + '?');
    if (/скло/.test(names))
      add('Яке скло?', 'Яке скло у годинника ' + n + '?');
    add('У чому переваги?', 'Які переваги годинника ' + n + ' можна підтвердити його характеристиками?');
    add('Кому підійде?', 'Для кого або яких ситуацій підійде годинник ' + n + ' з огляду на підтверджені характеристики?');
    add('З чим порівняти?', 'З якими схожими доступними моделями можна порівняти ' + n + '?');
    return items.slice(0, 6);
  }

  function addCardQuestions(product, card) {
    var all = makeWatchQuestions(product);    if (!all.length) return;
    var section = create('div', 'zorro-chat-questions');
    var title = create('div', 'zorro-chat-questions-label', 'Запитайте про цю модель');
    var list = create('div', 'zorro-chat-questions-list');
    section.appendChild(title);
    section.appendChild(list);
    function appendQuestion(question) {
      var button = create('button', 'zorro-chat-question', question.label);
      button.type = 'button';
      button.title = 'Запитати про ' + String(product.name || 'цей годинник');
      button.addEventListener('click', function () { sendMessage(question.text); });
      list.appendChild(button);
    }
    all.slice(0, 3).forEach(appendQuestion);
    if (all.length > 3) {
      var more = create('button', 'zorro-chat-more-questions', 'Ще запитання +');
      more.type = 'button';
      more.addEventListener('click', function () {
        all.slice(3).forEach(appendQuestion);
        more.parentNode.removeChild(more);
      });
      section.appendChild(more);
    }
    card.appendChild(section);
  }

    // Характеристики товару

    if (
      Array.isArray(product.attributes) &&
      product.attributes.length
    ) {
      var specs = create(
        'div',
        'zorro-chat-specs'
      );

              '',

            );

            specs.appendChild(specification);

      );

      card.appendChild(specs);
    }

    var actions = create(
      'div',
      'zorro-chat-card-actions'
    );

      '',

      '🔍 Схожі'
    );

          'Новий пошук: покажи схожі годинники до ' +

      }

    actions.appendChild(describe);
    actions.appendChild(similar);

    container.appendChild(card);

  // ВІДНОВЛЕННЯ ІСТОРІЇ

  function restoreChat() {
    while (messages.firstChild) {
      messages.removeChild(messages.firstChild);
    }

    if (!history.length) {
      showWelcome();
      return;
    }

    history.forEach(function (turn) {
      addBubble(
        turn.content,
        turn.role === 'user',
        false,
        knownProductLinks(turn.product_links, turn.products),
        turn.checkout
      );

      if (
        turn.role === 'assistant' &&
        Array.isArray(turn.products)
      ) {
        addProducts(turn.products);
      }
    });
  }

  try {
    var saved = JSON.parse(
      localStorage.getItem(STORAGE_KEY) || '{}'
    );

    if (
      saved &&
      typeof saved === 'object'
    ) {
      if (
        saved.state &&
        typeof saved.state === 'object'
      ) {
        state = saved.state;
      }

      if (Array.isArray(saved.history)) {
        history = saved.history.slice(-12);
      }
    }

  } catch (err) {
    history = [];
  }

  // ЗАПИТ ДО PYTHON-СЕРВЕРА
  // Сумісний з GTM: XMLHttpRequest замість async/await

  function request(path, payload, done) {
    if (!/^https:\/\//i.test(API)) {
      done(new Error('Вкажіть адресу Render'));
      return;
    }

    var xhr = new XMLHttpRequest();
    var completed = false;

    function finish(error, result) {
      if (completed) return;

      completed = true;
      done(error, result);
    }

    xhr.open('POST', API + path, true);
    xhr.timeout = 55000;

    xhr.setRequestHeader(
      'Content-Type',
      'application/json'
    );

    xhr.onreadystatechange = function () {
      if (xhr.readyState !== 4) return;

      if (
        xhr.status < 200 ||
        xhr.status >= 300
      ) {
        finish(
          new Error('HTTP ' + xhr.status)
        );
        return;
      }

      try {
        var data = JSON.parse(
          xhr.responseText
        );

        finish(null, data);

      } catch (err) {
        finish(err);
      }
    };

    xhr.onerror = function () {
      finish(new Error('Помилка мережі'));
    };

    xhr.ontimeout = function () {
      finish(
        new Error('Час очікування вичерпано')
      );
    };

    try {
      xhr.send(
        JSON.stringify(payload)
      );

    } catch (err) {
      finish(err);
    }
  }

  // СТАН ЗАВАНТАЖЕННЯ

  function setBusy(value) {
    busy = value;

    sendButton.disabled = value;
    input.disabled = value;

    if (value) {
      loading.style.display = 'block';
    } else {
      loading.style.display = 'none';
    }
  }

  // НАДСИЛАННЯ ПОВІДОМЛЕННЯ

  function sendMessage(value) {
    var text = String(
      value !== undefined ? value : input.value
    ).replace(/^\s+|\s+$/g, '');

    if (!text || busy) return;

    text = text.slice(0, 2000);

    // Попередню історію відправляємо до
    // додавання нового повідомлення.

    var previousHistory = history.slice(-12).map(
      function (turn) {
        return {
          role: turn.role,
          content: turn.content
        };
      }
    );

    setBusy(true);

    var quicks = messages.querySelectorAll(
      '.zorro-chat-quick'
    );

    for (var i = 0; i < quicks.length; i++) {
      quicks[i].style.display = 'none';
    }

    addBubble(text, true);
    input.value = '';
    if (isMobileChat() && document.activeElement === input) input.blur();

    request(
      '/api/chat',
      {
        message: text,
        history: previousHistory,
        state: state,
        current_product: currentProduct
      },
      function (error, data) {

        if (
          error ||
          !data ||
          typeof data.reply !== 'string'
        ) {
          console.warn(
            'Zorro AI:',
            error || 'Неправильна відповідь'
          );

          var errorBubble = addBubble(
            'Не вдалося з’єднатися з консультантом. ' +
            'Перевірте Render або спробуйте ще раз.',
            false,
            true
          );
          showAnswerStart(errorBubble);

        } else {

          var answerBubble = addBubble(
            data.reply,
            false,
            false,
            knownProductLinks(data.product_links, data.products),
            data.checkout
          );

          if (Array.isArray(data.products)) {
            addProducts(data.products);
          }

          if (
            data.state &&
            typeof data.state === 'object'
          ) {
            state = data.state;
          }

          history.push({
            role: 'user',
            content: text
          });

          history.push({
            role: 'assistant',
            content: data.reply,
            products: data.products || [],
            product_links: knownProductLinks(data.product_links, data.products),
            checkout: data.checkout || null
          });

          history = history.slice(-12);

          saveChat();
          showAnswerStart(answerBubble);
        }

        setBusy(false);
        // Не фокусуємо поле: на телефоні це повторно відкриває клавіатуру.
        onViewportChange();
      }
    );
  }

            if (result.status === 'success') {
              phoneInput.value = '';
              callback.style.display = 'none';
            }
          }

          callSend.disabled = false;
          setBusy(false);
        }
      );
    }
  );

  restoreChat();
